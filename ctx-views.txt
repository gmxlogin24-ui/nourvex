<script>
(function () {
    const recipientInput = document.getElementById('recipientInput');
    const amountInput = document.getElementById('amountInput');
    const noteInput = document.getElementById('noteInput');
    const resolveStatus = document.getElementById('resolveStatus');
    const continueBtn = document.getElementById('continueBtn');
    const hasPin = {{ ($hasPin ?? false) ? 'true' : 'false' }};
    let resolveTimer = null;
    let resolvedName = null;

    function tryResolve() {
        continueBtn.disabled = true;
        resolvedName = null;
        resolveStatus.textContent = '';

        const identifier = recipientInput.value.trim();
        if (!identifier) return;

        resolveStatus.textContent = 'Looking up recipient...';
        resolveStatus.className = 'text-sm mb-3 text-muted';

        fetch('{{ route("wallet.send.resolve") }}', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]')?.content || '{{ csrf_token() }}',
                'Accept': 'application/json',
            },
            body: JSON.stringify({ recipient_identifier: identifier }),
        })
        .then(res => res.json().then(data => ({ ok: res.ok, data })))
        .then(({ ok, data }) => {
            if (!ok) {
                resolveStatus.textContent = data.error || 'Could not find recipient.';
                resolveStatus.className = 'text-sm mb-3 text-red-500';
                return;
            }
            resolvedName = data.name;
            resolveStatus.textContent = 'Sending to: ' + data.name;
            resolveStatus.className = 'text-sm mb-3 text-accent';
            maybeEnableContinue();
        })
        .catch(() => {
            resolveStatus.textContent = 'Could not verify recipient. Check your connection and try again.';
            resolveStatus.className = 'text-sm mb-3 text-red-500';
        });
    }

    function maybeEnableContinue() {
        continueBtn.disabled = !(hasPin && resolvedName && amountInput.value && Number(amountInput.value) > 0);
    }

    function debounceResolve() {
        clearTimeout(resolveTimer);
        resolveTimer = setTimeout(tryResolve, 500);
    }

    recipientInput.addEventListener('input', debounceResolve);
    amountInput.addEventListener('input', maybeEnableContinue);

    // ---- Step transition ----
    const detailsStep = document.getElementById('detailsStep');
    const pinStep = document.getElementById('pinStep');
    const pinAmountLabel = document.getElementById('pinAmountLabel');
    const pinRecipientLabel = document.getElementById('pinRecipientLabel');
    const pinError = document.getElementById('pinError');
    const dots = document.querySelectorAll('.pin-dot');
    let pin = '';

    continueBtn.addEventListener('click', function () {
        pinAmountLabel.textContent = '₦' + Number(amountInput.value).toLocaleString();
        pinRecipientLabel.textContent = 'to ' + resolvedName;
        detailsStep.style.display = 'none';
        pinStep.style.display = 'block';
    });

    document.getElementById('backBtn').addEventListener('click', function () {
        pin = '';
        renderDots();
        pinError.textContent = '';
        pinStep.style.display = 'none';
        detailsStep.style.display = 'block';
    });

    function renderDots() {
        dots.forEach((d, i) => d.classList.toggle('filled', i < pin.length));
    }

    document.querySelectorAll('.pin-key').forEach(function (btn) {
        btn.addEventListener('click', function () {
            if (pin.length >= 6) return;
            pin += btn.dataset.key;
            renderDots();
            if (pin.length === 6) submitTransfer();
        });
    });

    document.getElementById('pinBackspace').addEventListener('click', function () {
        pin = pin.slice(0, -1);
        renderDots();
    });

    function submitTransfer() {
        if (!confirm('Send this money? Wallet transfers cannot be reversed.')) {
            pin = '';
            renderDots();
            return;
        }
        document.getElementById('hiddenRecipient').value = recipientInput.value.trim();
        document.getElementById('hiddenAmount').value = amountInput.value;
        document.getElementById('hiddenNote').value = noteInput.value;
        document.getElementById('hiddenPin').value = pin;
        document.getElementById('sendMoneyForm').submit();
    }
})();
</script>
@endsection

<script>
(function(){
    var folder = document.getElementById('balanceFolder');
    var slot = document.getElementById('folderCardSlot');
    if (!folder || !slot) return;

    var tapTimer = null, tapCount = 0;

    function openFull(){ folder.classList.remove('is-peek'); folder.classList.add('is-open'); }
    function peek(){ folder.classList.toggle('is-peek'); }
    function closeAll(){ folder.classList.remove('is-open','is-peek'); }

    function handleTap(e){
        var link = e.target.closest('a,button');
        if (link) {
            if (!folder.classList.contains('is-open')) { e.preventDefault(); }
            else { return; }
        }
        tapCount++;
        if (tapCount === 1) {
            tapTimer = setTimeout(function(){
                tapCount = 0;
                folder.classList.contains('is-open') ? closeAll() : peek();
            }, 280);
        } else if (tapCount === 2) {
            clearTimeout(tapTimer);
            tapCount = 0;
            folder.classList.contains('is-open') ? closeAll() : openFull();
        }
    }

    slot.addEventListener('click', handleTap);
    document.addEventListener('click', function(e){
        if (folder.classList.contains('is-open') && !folder.contains(e.target)) closeAll();
    });
})();
</script>

    public function revenue()
    {
        $this->guard();

        $totalAllTime = PlatformRevenue::sum('amount');
        $totalThisMonth = PlatformRevenue::where('created_at', '>=', now()->startOfMonth())->sum('amount');
        $totalThisWeek = PlatformRevenue::where('created_at', '>=', now()->startOfWeek())->sum('amount');
        $totalToday = PlatformRevenue::where('created_at', '>=', now()->startOfDay())->sum('amount');

        $bySource = PlatformRevenue::selectRaw('source, SUM(amount) as total, COUNT(*) as count')
            ->groupBy('source')
            ->orderByDesc('total')
            ->get();

        $recent = PlatformRevenue::latest()->take(50)->get();

        return view('admin.revenue', [
            'totalAllTime' => $totalAllTime,
            'totalThisMonth' => $totalThisMonth,
            'totalThisWeek' => $totalThisWeek,
            'totalToday' => $totalToday,
            'bySource' => $bySource,
            'recent' => $recent,
        ]);
    }

    private function guard()
    {
        abort_unless(Auth::check() && Auth::user()->is_admin, 403);
    }

    public function index()
    {
        $this->guard();
        $subStatusCounts = Subscription::selectRaw('status, COUNT(*) as total')->groupBy('status')->pluck('total', 'status');
        $subMethodRevenue = Subscription::where('status', 'active')
            ->selectRaw('payment_method, SUM(amount) as total')
            ->groupBy('payment_method')
            ->pluck('total', 'payment_method');

        return view('admin.index', [
            'userCount' => User::count(),
            'bannedCount' => User::where('is_banned', true)->count(),
            'tierCounts' => User::selectRaw('tier, COUNT(*) as total')->groupBy('tier')->pluck('total', 'tier'),

            'blogCount' => Blog::count(),
            'recentUsers' => User::latest()->take(8)->get(),
            'recentTx' => Transaction::latest()->take(10)->get(),
            'totalRevenue' => Transaction::whereIn('type', ['tier_upgrade'])->sum('amount') * -1,
            'subActive' => $subStatusCounts['active'] ?? 0,
            'subPending' => $subStatusCounts['pending_payment'] ?? 0,

                ->with('user')->orderBy('expires_at')->take(10)->get(),

    // ===== USERS =====
    public function users(Request $r)
    {
        $this->guard();
        $q = User::query();
        if ($r->filled('search')) {
            $q->where(fn($w) => $w->where('name', 'like', "%{$r->search}%")->orWhere('email', 'like', "%{$r->search}%"));
        }
        if ($r->filled('tier')) $q->where('tier', $r->tier);
        if ($r->filled('banned')) $q->where('is_banned', $r->banned === '1');

        return view('admin.users', ['users' => $q->latest()->paginate(25)->withQueryString()]);
    }

    public function grantAccess(Request $r)
    {
        $this->guard();
        $users = collect();
        if ($r->filled('search')) {
            $users = User::where(fn($w) => $w->where('name', 'like', "%{$r->search}%")->orWhere('email', 'like', "%{$r->search}%"))
                ->limit(20)->get();
        }
        return view('admin.grant-access', ['users' => $users, 'search' => $r->search]);
    }

    public function grantAccessStore(Request $r)
    {
        $this->guard();
        $validated = $r->validate([
            'user_id' => 'required|exists:users,id',
            'tier' => 'required|in:tier1,tier2,tier3',
        ]);

        $user = User::findOrFail($validated['user_id']);
        $user->update([
            'signup_fee_paid' => true,
            'badge_tier' => 'bronze',
            'tier' => $validated['tier'],
            'tier_upgraded_at' => now(),
        ]);

        return back()->with('status', "Granted {$validated['tier']} access to {$user->name}.");
    }

    public function toggleAdmin(User $user)
    {
        $this->guard();
        abort_if($user->id === Auth::id(), 403, 'You cannot remove your own admin access.');
        $user->update(['is_admin' => !$user->is_admin]);
        return back()->with('status', $user->name . ' admin status updated.');
    }

    public function toggleBan(User $user)
    {
        $this->guard();
        abort_if($user->id === Auth::id(), 403, 'You cannot ban yourself.');
        $user->update(['is_banned' => !$user->is_banned]);
        if ($user->is_banned) {
            $user->notify(new \App\Notifications\AccountBanned());
        }
        return back()->with('status', $user->name . ($user->is_banned ? ' banned.' : ' unbanned.'));
    }

    // Flips a user in/out of the customer_care role. Does not touch is_admin -
    // customer_care is a separate, lesser-privileged role, not an admin variant.
    public function toggleCustomerCare(User $user)
    {
        $this->guard();
        abort_if($user->id === Auth::id(), 403, 'You cannot change your own role.');
        abort_if($user->isAdmin(), 403, 'Admins are not eligible for the customer_care role.');

        $newRole = $user->role === 'customer_care' ? 'user' : 'customer_care';
        $user->update(['role' => $newRole]);

        return back()->with('status', $user->name . ' is now ' . ($newRole === 'customer_care' ? 'Customer Care.' : 'a regular user.'));
    }

    }

    // Locked to prevent this admin write from racing with a concurrent user-side wallet
    // operation (topup, purchase, gig payout) on the same account and losing an update.
    public function adjustBalance(Request $r, User $user)
    {
        $this->guard();
        $data = $r->validate(['amount' => 'required|numeric', 'note' => 'nullable|string|max:255']);

        DB::transaction(function () use ($data, $user) {
            $lockedUser = User::where('id', $user->id)->lockForUpdate()->first();
            $lockedUser->increment('wallet_balance', $data['amount']);
            Transaction::create([
                'user_id' => $lockedUser->id,
                'type' => 'admin_adjustment',
                'amount' => $data['amount'],
                'description' => $data['note'] ?? 'Admin balance adjustment',
            ]);
            $lockedUser->notify(new \App\Notifications\AdminBalanceAdjusted($data['amount'], $data['note'] ?? null));
        });

        return back()->with('status', 'Balance adjusted for ' . $user->name);
    }

    // ===== GIGS =====
    public function gigs(Request $r)
    {
        $this->guard();
        $q = Gig::with('user')->latest();
        if ($r->filled('status')) $q->where('status', $r->status);
        return view('admin.gigs', ['gigs' => $q->paginate(25)->withQueryString()]);
    }

    public function deleteGig(Gig $gig)
    {
        $this->guard();
        $gig->delete();
        return back()->with('status', 'Gig removed.');
    }

    // ===== LISTINGS =====
    public function listings(Request $r)
    {

        if ($r->filled('status')) $q->where('status', $r->status);
        return view('admin.listings', ['listings' => $q->paginate(25)->withQueryString()]);

    }

    // ===== TRANSACTIONS =====
    public function transactions(Request $r)
    {
        $this->guard();
        $q = Transaction::with('user')->latest();
        if ($r->filled('type')) $q->where('type', $r->type);
        if ($r->filled('search')) {
            $q->whereHas('user', fn($w) => $w->where('name', 'like', "%{$r->search}%"));
        }
        return view('admin.transactions', ['transactions' => $q->paginate(30)->withQueryString()]);
    }

    // ===== SUBSCRIPTIONS =====
    public function subscriptions(Request $r)

<?php

namespace App\Providers;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\URL;
use Illuminate\Support\Facades\View;
use Illuminate\Support\Facades\Event;
use App\Events\TierUpgraded;
use App\Listeners\RecordTierUpgradeReveal;

class AppServiceProvider extends ServiceProvider
{
    /**
     * Register any application services.
     */
    public function register(): void
    {
        //
    }

    /**
     * Bootstrap any application services.
     */
    public function boot(): void
    {
        if (request()->header('X-Forwarded-Proto') === 'https' || request()->secure()) {
            URL::forceScheme('https');
        }

        Event::listen(TierUpgraded::class, RecordTierUpgradeReveal::class);

        View::composer('*', function ($view) {
            if (auth()->check()) {
                $user = auth()->user();
                $view->with('navAvatar', $user->avatar_url);
                $view->with('navInitial', strtoupper(substr($user->name ?? 'U', 0, 1)));
                $view->with('navQCoinBalance', $user->q_coin_balance ?? 0);
                $unreadChats = \App\Models\ConversationParticipant::where('user_id', $user->id)
                    ->whereHas('conversation', fn($q) => $q->where('type', 'business'))
                    ->with('conversation')
                    ->get()
                    ->sum(fn($p) => $p->conversation->unreadCountFor($user->id));
                $view->with('navUnreadChats', $unreadChats);
                $view->with('navUnreadNotifications', $user->unreadNotifications()->count());
            }
        });
    }
}
