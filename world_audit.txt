<?php

namespace App\Services;

use App\Models\RankPoint;
use App\Models\User;
use Illuminate\Database\UniqueConstraintViolationException;

class RankingService
{
    // $ref makes an award one-time (e.g. "login:2026-10-03"). Same ref twice = ignored.
    public function award(User $user, string $source, ?int $points = null, ?string $ref = null): ?RankPoint
    {
        $base = $points ?? (int) config("ranking.points.$source", 0);
        if ($base <= 0) {
            return null;
        }

        if ($ref !== null && RankPoint::where('user_id', $user->id)->where('source', $source)->where('ref', $ref)->exists()) {
            return null;
        }

        $mult = (float) config('ranking.plan_multiplier.' . $user->planKey(), 1);

        try {
            return RankPoint::create([
                'user_id' => $user->id,
                'source' => $source,
                'points' => (int) round($base * $mult),
                'ref' => $ref,
            ]);
        } catch (UniqueConstraintViolationException $e) {
            return null;
        }
    }

    public function since(string $period)
    {
        return match ($period) {
            'week' => now()->startOfWeek(),
            'month' => now()->startOfMonth(),
            default => null,
        };
    }

    public function total(User $user, string $period = 'all'): int
    {
        $q = RankPoint::where('user_id', $user->id);
        if ($since = $this->since($period)) {
            $q->where('created_at', '>=', $since);
        }
        return (int) $q->sum('points');
    }

    // $period: week | month | all. $area: null = everyone.
    public function board(string $period = 'week', ?string $area = null, int $limit = 20)
    {
        $q = RankPoint::query()
            ->join('users', 'users.id', '=', 'rank_points.user_id')
            ->where('users.leaderboard_opt_in', true)
            ->selectRaw('users.id, users.name, users.avatar_url, users.area, users.tier, SUM(rank_points.points) as total')
            ->groupBy('users.id', 'users.name', 'users.avatar_url', 'users.area', 'users.tier')
            ->orderByDesc('total')
            ->limit($limit);

        if ($since = $this->since($period)) {
            $q->where('rank_points.created_at', '>=', $since);
        }
        if ($area) {
            $q->where('users.area', $area);
        }

        return $q->get();
    }
}

namespace App\Http\Controllers;

use App\Models\Conversation;
use App\Services\Concierge\KeywordConciergeService;
use App\Services\TierService;
use Illuminate\Http\Request;
use RuntimeException;

class ConciergeController extends Controller
{
    public function show(TierService $tiers)
    {
        $user = auth()->user();
        abort_unless($tiers->unlocked($user, 'concierge.actions'), 403, 'Upgrade to unlock the QLEVO Concierge.');

        $conversation = Conversation::findOrCreateConcierge($user);
        return view('concierge.index', ['messages' => $conversation->messages()->get()]);
    }

    public function sendMessage(Request $request, KeywordConciergeService $concierge, TierService $tiers)
    {
        $user = auth()->user();
        if (!$tiers->unlocked($user, 'concierge.actions')) {
            return response()->json(['error' => 'Upgrade to unlock the QLEVO Concierge.'], 403);
        }

        $data = $request->validate(['message' => 'required|string|max:2000']);
        $conversation = Conversation::findOrCreateConcierge($user);

        \App\Models\ConversationMessage::create([
            'conversation_id' => $conversation->id,
            'sender_id' => $user->id,
            'message_type' => 'text',
            'body' => $data['message'],
        ]);

        $result = $concierge->handleUserMessage($user, $data['message']);

        \App\Models\ConversationMessage::create([
            'conversation_id' => $conversation->id,
            'is_ai' => true,
            'message_type' => 'text',
            'body' => $result['reply'],
        ]);
        $conversation->update(['last_message_at' => now()]);

        return response()->json($result);
    }
}

(function () {
    var slides = document.querySelectorAll('#mgHero .mg-slide');
    var dots = document.querySelectorAll('#mgHero .mg-dot');
    var current = 0;
    window.mgShowSlide = function (i) {
        slides[current].classList.remove('active');
        dots[current].classList.remove('active');
        current = i;
        slides[current].classList.add('active');
        dots[current].classList.add('active');
    };
    if (slides.length > 1) {
        setInterval(function () { mgShowSlide((current + 1) % slides.length); }, 4500);
    }

    // ── Featured Products rotating carousel ─────────────────────────
    // pool[0..n] is the ordered list of products. Card i always shows
    // pool[i]. Self-rotate moves every item left by one (wraps around) —
    // no new content, just reshuffles what's already there, every 15s.
    // Adding a real new product unshifts it to pool[0] and every card
    // cascades down one slot, exactly like a news ticker.
    var pool = @json(array_values($featuredPool));
    var cardEls = Array.from(document.querySelectorAll('#mgFeaturedGrid .mg-flip-scene'));

    function mgRenderFaceHTML(item) {
        var oldPriceHtml = item.oldPrice ? '<span class="mg-pcard-old">₦' + Number(item.oldPrice).toLocaleString() + '</span>' : '';
        return '<div class="mg-pcard-img">' +
                    '<span class="mg-pcard-badge">' + item.badge + '</span>' +
                    '<span class="mg-pcard-heart"><i data-lucide="heart" class="w-3.5 h-3.5"></i></span>' +
                    '<img src="https://picsum.photos/seed/' + item.seed + '/300/300" alt="' + item.title + '">' +
                '</div>' +
                '<div class="mg-pcard-body">' +
                    '<p class="mg-pcard-title">' + item.title + '</p>' +
                    '<p><span class="mg-pcard-price">₦' + Number(item.price).toLocaleString() + '</span>' + oldPriceHtml + '</p>' +
                '</div>';
    }

    function mgUpdateCard(scene, item) {
        var inner = scene.querySelector('.mg-flip-inner');
        var isFlipped = inner.classList.contains('flipped');
        var hiddenFace = isFlipped ? scene.querySelector('.mg-pcard-face-a') : scene.querySelector('.mg-pcard-face-b');
        hiddenFace.innerHTML = mgRenderFaceHTML(item);
        scene.classList.add('flipping');
        inner.classList.toggle('flipped');
        setTimeout(function () { scene.classList.remove('flipping'); if (window.lucide) lucide.createIcons(); }, 1000);
    }

    function mgCascadeRender(staggerMs) {
        cardEls.forEach(function (scene, i) {
            var item = pool[i];
            if (!item) return;
            setTimeout(function () { mgUpdateCard(scene, item); }, i * staggerMs);
        });
    }

    // Manual tap: advance just that one card to the next item in the pool.
    window.mgManualAdvance = function (scene) {
        var i = cardEls.indexOf(scene);
        if (i === -1) return;
        var nextItem = pool[(i + 1) % pool.length];
        mgUpdateCard(scene, nextItem);
    };

    // Ready for real backend integration: call this whenever a genuinely
    // new featured product is created. Example:
    //   mgAddFeaturedProduct({badge:'NEW', title:'...', price:1000, oldPrice:null, seed:'xyz'});
    window.mgAddFeaturedProduct = function (item) {
        pool.unshift(item);
        if (pool.length > cardEls.length) pool.pop();
        mgCascadeRender(250);
    };

    function mgSelfRotate() {
        pool.push(pool.shift());
        mgCascadeRender(250);
    }

    if (cardEls.length > 1) {
        setInterval(mgSelfRotate, 15000);
    }
})();
</script>

<?php

namespace App\Services;

use App\Models\RankPoint;
use App\Models\User;
use Illuminate\Database\UniqueConstraintViolationException;

class RankingService
{
    // $ref makes an award one-time (e.g. "login:2026-10-03"). Same ref twice = ignored.
    public function award(User $user, string $source, ?int $points = null, ?string $ref = null): ?RankPoint
    {
        $base = $points ?? (int) config("ranking.points.$source", 0);
        if ($base <= 0) {
            return null;
        }

        if ($ref !== null && RankPoint::where('user_id', $user->id)->where('source', $source)->where('ref', $ref)->exists()) {
            return null;
        }

        $mult = (float) config('ranking.plan_multiplier.' . $user->planKey(), 1);

        try {
            return RankPoint::create([
                'user_id' => $user->id,
                'source' => $source,
                'points' => (int) round($base * $mult),
                'ref' => $ref,
            ]);
        } catch (UniqueConstraintViolationException $e) {
            return null;
        }
    }

    public function since(string $period)
    {
        return match ($period) {
            'week' => now()->startOfWeek(),
            'month' => now()->startOfMonth(),
            default => null,
        };
    }

    public function total(User $user, string $period = 'all'): int
    {
        $q = RankPoint::where('user_id', $user->id);
        if ($since = $this->since($period)) {
            $q->where('created_at', '>=', $since);
        }
        return (int) $q->sum('points');
    }

    // $period: week | month | all. $area: null = everyone.
    public function board(string $period = 'week', ?string $area = null, int $limit = 20)
    {
        $q = RankPoint::query()
            ->join('users', 'users.id', '=', 'rank_points.user_id')
            ->where('users.leaderboard_opt_in', true)
            ->selectRaw('users.id, users.name, users.avatar_url, users.area, users.tier, SUM(rank_points.points) as total')
            ->groupBy('users.id', 'users.name', 'users.avatar_url', 'users.area', 'users.tier')
            ->orderByDesc('total')
            ->limit($limit);

        if ($since = $this->since($period)) {
            $q->where('rank_points.created_at', '>=', $since);
        }
        if ($area) {
            $q->where('users.area', $area);
        }

        return $q->get();
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Transaction;
use App\Services\RankingService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class LeaderboardController extends Controller
{
    public function index(Request $request, RankingService $ranking)
    {
        $user = Auth::user();
        $areas = config('areas.list');

        $type = $request->query('type') === 'earnings' ? 'earnings' : 'points';
        $period = in_array($request->query('period'), ['week', 'month', 'all'], true) ? $request->query('period') : 'week';
        $area = array_key_exists((string) $request->query('area'), $areas) ? $request->query('area') : null;

        if ($type === 'earnings') {
            $q = Transaction::query()
                ->where('transactions.amount', '>', 0)
                ->whereIn('transactions.type', ['gig_earning', 'marketplace_sale', 'referral_bonus'])
                ->join('users', 'transactions.user_id', '=', 'users.id')
                ->where('users.leaderboard_opt_in', true)
                ->selectRaw('users.id, users.name, users.area, SUM(transactions.amount) as total')
                ->groupBy('users.id', 'users.name', 'users.area')
                ->orderByDesc('total')
                ->take(50);

            if ($since = $ranking->since($period)) {
                $q->where('transactions.created_at', '>=', $since);
            }
            if ($area) {
                $q->where('users.area', $area);
            }
            $rows = $q->get();
        } else {
            $rows = $ranking->board($period, $area, 50);
        }

        $myRank = null;
        foreach ($rows as $i => $row) {
            if ((int) $row->id === (int) $user->id) {
                $myRank = $i + 1;
                break;
            }
        }

        return view('leaderboard.index', [
            'rows' => $rows,
            'type' => $type,
            'period' => $period,
            'area' => $area,
            'areas' => $areas,
            'optedIn' => $user->leaderboard_opt_in,
            'myPoints' => $ranking->total($user, $period),
            'myRank' => $myRank,
        ]);
    }

    public function toggleOptIn(Request $request)
    {
        $user = Auth::user();
        $user->update(['leaderboard_opt_in' => !$user->leaderboard_opt_in]);

        return back()->with('status', $user->leaderboard_opt_in ? 'You\'re now on the leaderboard!' : 'You\'ve been removed from the leaderboard.');
    }
}
