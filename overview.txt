<?php

namespace App\Http\Middleware;

use App\Models\Listing;
use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class EnsurePlanAllows
{
    protected array $messages = [
        'gigs_post' => 'Posting gigs needs the Gold plan or above.',
        'market_sell' => 'Selling on the marketplace needs the Silver plan or above.',
        'market_buy' => 'Buying on the marketplace needs the Silver plan or above.',
    ];

    // Usage: ->middleware('plan:gigs_post')
    public function handle(Request $request, Closure $next, string $rule)
    {
        $user = Auth::user();
        if (!$user) {
            return redirect()->route('login');
        }

        if (!$user->planAllows($rule)) {
            return redirect()->route('subscriptions.index')->withErrors([
                'tier' => $this->messages[$rule] ?? 'Upgrade your plan to do this.',
            ]);
        }

        if ($rule === 'market_sell' && $request->isMethod('post') && !$user->isAdmin()) {
            $cap = $user->planRule('market_weekly_cap');
            if ($cap !== null) {
                $used = Listing::where('user_id', $user->id)->where('created_at', '>=', now()->startOfWeek())->count();
                if ($used >= $cap) {
                    return back()->withErrors([
                        'tier' => 'You have reached your weekly listing limit (' . $cap . ') on the ' . $user->planLabel() . ' plan. Upgrade for a higher limit.',
                    ]);
                }
            }
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use App\Models\Listing;

class RequireTier
{
    public function handle(Request $request, Closure $next)
    {
        $user = Auth::user();

        // Admins are never gated by tier restrictions.
        if ($user->is_admin) {
            return $next($request);
        }

        // No free tier - everyone must pay (vendor registration) to reach tier1 before posting.
        if ($user->tier === 'free') {
            return redirect()->route('tier.index')->withErrors([
                'tier' => 'You need to upgrade to Tier 1 to post gigs or marketplace listings.'
            ]);
        }

        // Only check the limit on the actual submit (POST), not the form page (GET).
        // Gigs are no longer capped here - the cap on gigs is how many a user can
        // have *claimed* at once (see GigController::claimCap()), not how many
        // they post. Listings are still capped on posting.
        if ($request->isMethod('post') && $user->tier === 'tier1') {
            $activeListings = Listing::where('user_id', $user->id)->count();

            if ($activeListings >= 5) {
                return back()->withErrors([
                    'tier' => 'Tier 1 allows up to 5 active listings. Upgrade to Tier 2 for unlimited listings.'
                ]);
            }
        }

        return $next($request);
    }
}

<?php

namespace App\Models;

// use Illuminate\Contracts\Auth\MustVerifyEmail;
use Database\Factories\UserFactory;
use Illuminate\Database\Eloquent\Attributes\Fillable;
use Illuminate\Database\Eloquent\Attributes\Hidden;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use NotificationChannels\WebPush\HasPushSubscriptions;
use Illuminate\Support\Facades\Hash;
use App\Events\TierUpgraded;

#[Fillable([
    'name', 'email', 'password', 'gender', 'spoken_language', 'avatar_url', 'background_url', 'accent_color',
    'referral_code', 'referred_by', 'wallet_balance', 'q_coin_balance', 'q_coin_last_claim_date',
    'tier', 'tier_upgraded_at', 'is_verified_seller',
    'virtual_account_number', 'virtual_account_bank', 'virtual_account_ref',
    'account_type', 'business_name', 'business_logo', 'business_whatsapp', 'is_vendor_active', 'interests',
    'streak_days', 'last_active_date', 'leaderboard_opt_in', 'q_coin_streak_days', 'wallet_restricted', 'google_id', 'transaction_pin', 'theme_style', 'mobile_nav_style', 'sidebar_style',

    'signup_fee_paid', 'badge_tier', 'card_token', 'card_last4', 'card_type',
    'is_banned', 'subscription_expires_at', 'skills',
    'notif_style', 'navbar_style', 'topbar_style', 'card_style', 'area', 'area_verified_at',

#[Hidden(['password', 'remember_token'])]
class User extends Authenticatable
{
    /** @use HasFactory<UserFactory> */
    use HasFactory, Notifiable, HasPushSubscriptions, HasPlanRules;

    protected function casts(): array
    {
        return [
            'email_verified_at' => 'datetime',
            'password' => 'hashed',
            'transaction_pin' => 'hashed',
            'interests' => 'array',
            'customer_care_permissions' => 'array',
            'is_admin' => 'boolean',
            'leaderboard_opt_in' => 'boolean',
            'wallet_restricted' => 'boolean',
            'is_vendor_active' => 'boolean',
            'tier_upgraded_at' => 'datetime', 'area_verified_at' => 'datetime',
        ];
    }

    protected static function booted(): void
    {
        static::updated(function (User $user) {
            if ($user->wasChanged('tier')) {
                event(new TierUpgraded($user, $user->getOriginal('tier'), $user->tier));
            }
        });
    }

    /**
     * Get the attributes that should be cast.
     *
     * @return array<string, string>
     */
    public function completedGigsCount(): int
    {
        return \App\Models\Gig::where('accepted_by', $this->id)->where('status', 'completed')->count();
    }

    public function favoriteListings()
    {
        return $this->belongsToMany(Listing::class, 'listing_favorites')->withTimestamps();
    }

    /**
     * User's level relationship
     */
    public function level()
    {
        return $this->belongsTo(Level::class);
    }

    /**
     * User's badges relationship
     */
    public function badges()
    {
        return $this->belongsToMany(Badge::class, 'user_badges')
            ->withPivot('earned_at')
            ->withTimestamps();
    }

    public function averageRatingReceived(): ?float
    {
        $avg = \App\Models\Review::where('reviewed_user_id', $this->id)->avg('rating');
        return $avg ? round($avg, 1) : null;
    }

    public function reviewsReceivedCount(): int
    {
        return \App\Models\Review::where('reviewed_user_id', $this->id)->count();
    }

    // Fiverr-style progression, adapted: New -> Rising Talent -> Level 2 -> Top Rated.
    // Ratings gate the higher tiers so volume alone can't buy the top badge.
    public function workerLevel(): array
    {
        $completed = $this->completedGigsCount();
        $rating = $this->averageRatingReceived();

        if ($completed >= 20 && $rating !== null && $rating >= 4.5) {
            return ['key' => 'top_rated', 'label' => 'Top Rated', 'icon' => 'trophy'];
        }
        if ($completed >= 5 && ($rating === null || $rating >= 4.0)) {
            return ['key' => 'level_2', 'label' => 'Level 2', 'icon' => 'star'];
        }
        if ($completed >= 1) {
            return ['key' => 'rising', 'label' => 'Rising Talent', 'icon' => 'trending-up'];
        }
        return ['key' => 'new', 'label' => 'New', 'icon' => 'sprout'];
    }

    // Transaction PIN gates real-money purchases (VTU, cable, withdrawals).
    // Hashed via the cast below, same treatment as password - never compared
    // or stored in plaintext.
    // Users without a verified virtual account (BVN) are capped on how much
    // they can hold in-wallet - defends against using the wallet to stash/move
    // money with zero identity verification. Earnings owed to the user (gigs,
    // marketplace sales, referral bonuses) are NOT subject to this cap - only
    // top-ups and incoming transfers, which are funds the user chooses to add.
    const NO_BVN_WALLET_CAP = 20000;

    public function wouldExceedNoBvnCap(float $incomingAmount): bool
    {
        if ($this->virtual_account_number) {
            return false;
        }
        return ($this->wallet_balance + $incomingAmount) > self::NO_BVN_WALLET_CAP;
    }

    public function hasTransactionPin(): bool
    {
        return !empty($this->transaction_pin);
    }

    public function verifyTransactionPin(string $pin): bool
    {
        return Hash::check($pin, $this->transaction_pin ?? '');
    }

    public function isAdmin(): bool
    {
        return $this->is_admin || $this->role === 'admin';
    }

    public function isCustomerCare(): bool
    {
        return $this->role === 'customer_care';
    }

    public function isCustomerCareOrAdmin(): bool
    {
        return $this->isAdmin() || $this->isCustomerCare();
    }

    public function hasCcPermission(string $key): bool
    {
        if ($this->isAdmin()) {
            return true;
        }
        return in_array($key, $this->customer_care_permissions ?? [], true);
    }
}

=== PaystackWebhookController ===
<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use App\Models\User;
use App\Models\Transaction;
use App\Models\WalletTopup;
use App\Models\Withdrawal;
use App\Models\Subscription;

class PaystackWebhookController extends Controller
{
    public function handle(Request $request)
    {
        $signature = $request->header('x-paystack-signature') ?? '';
        $secret = config('services.paystack.secret') ?? '';
        $expected = hash_hmac('sha512', $request->getContent(), $secret);

        if ($secret === '' || !hash_equals($expected, $signature)) {
            return response('Unauthorized', 401);
        }

        $payload = $request->all();
        $event = $payload['event'] ?? '';

        if ($event === 'transfer.success' || $event === 'transfer.failed') {
            $this->handleTransferCompleted($payload['data'] ?? [], $event === 'transfer.success');
            return response('OK', 200);
        }

        if ($event !== 'charge.success') {
            return response('Ignored', 200);
        }

        $data = $payload['data'] ?? [];

        if (($data['status'] ?? '') !== 'success') {
            return response('Not successful', 200);
        }
        if (($data['currency'] ?? null) !== 'NGN') {
            return response('Currency mismatch', 200);
        }

        $paystackId = $data['id'] ?? null;
        $amount = (float) ($data['amount'] ?? 0) / 100;

        if (!$paystackId || $amount <= 0) {
            return response('Missing data', 200);
        }

        $txRef = $data['reference'] ?? null;

        if ($txRef && str_starts_with($txRef, 'SUBSCRIBE-')) {
            DB::transaction(function () use ($txRef, $amount) {
                $sub = Subscription::where('paystack_tx_ref', $txRef)->lockForUpdate()->first();
                if ($sub && $sub->status === 'pending_payment' && $amount >= (float) $sub->amount) {
                    $sub->activate();
                }
            });
            return response('OK', 200);
        }

        if ($txRef && str_starts_with($txRef, 'QELVO-')) {
            $this->settleTopupByRef($txRef, $amount);
            return response('OK', 200);
        }

        if ($txRef && str_starts_with($txRef, 'VENDORREG-')) {
            // Vendor registration is settled by PaystackController@vendorCallback - never a wallet top-up.
            return response('OK', 200);
        }

        if ($txRef && str_starts_with($txRef, 'SIGNUPFEE-')) {
            $feeTopup = WalletTopup::where('tx_ref', $txRef)->first();
            if ($feeTopup && $amount >= (float) $feeTopup->amount) {
                DB::transaction(function () use ($feeTopup, $amount, $data) {
                    $lockedTopup = WalletTopup::where('id', $feeTopup->id)->lockForUpdate()->first();
                    if ($lockedTopup->status === 'successful') return;

                    $user = User::where('id', $lockedTopup->user_id)->lockForUpdate()->first();
                    if (!$user) return;

                    $lockedTopup->update(['status' => 'successful']);
                    if ($user->signup_fee_paid) return;

                    $update = ['signup_fee_paid' => true, 'badge_tier' => 'bronze'];
                    if ($user->tier === 'free') {
                        $update['tier'] = 'tier1';
                        $update['tier_upgraded_at'] = now();
                    }
                    $auth = $data['authorization'] ?? null;
                    if ($auth && !empty($auth['authorization_code'])) {
                        $update['card_token'] = $auth['authorization_code'];
                        $update['card_last4'] = $auth['last4'] ?? null;
                        $update['card_type'] = $auth['card_type'] ?? null;
                    }
                    $user->update($update);

                    Transaction::create(['user_id' => $user->id, 'type' => 'signup_fee', 'amount' => -$amount, 'description' => 'Account activation via Paystack']);
                });
            }
            return response('OK', 200);
        }

        // Dedicated Virtual Account credit
        $accountNumber = $data['authorization']['receiver_bank_account_number']
            ?? $data['metadata']['receiver_account_number']
            ?? null;

        $this->creditWallet((string) $paystackId, $amount, $accountNumber, $data['customer']['email'] ?? null);

        return response('OK', 200);
    }

    private function handleTransferCompleted(array $data, bool $successful): void
    {
        $reference = $data['reference'] ?? null;
        if (!$reference) return;

        if ($successful) {
            DB::transaction(function () use ($reference) {
                $withdrawal = Withdrawal::where('tx_ref', $reference)->lockForUpdate()->first();
                if ($withdrawal && $withdrawal->status === 'pending') {
                    $withdrawal->update(['status' => 'successful']);
                }
            });
            return;
        }

        app(WalletActionController::class)->reverseWithdrawal($reference, 'Withdrawal failed - refunded to wallet');
    }

    private function settleTopupByRef(string $txRef, float $amount): void
    {
        $topup = WalletTopup::where('tx_ref', $txRef)->first();
        if (!$topup) return;

        $notifyUser = null;

        DB::transaction(function () use ($topup, $amount, &$notifyUser) {
            $lockedTopup = WalletTopup::where('id', $topup->id)->lockForUpdate()->first();
            if ($lockedTopup->status === 'successful') return;
            if ($amount < (float) $lockedTopup->amount) return;

            $user = User::where('id', $lockedTopup->user_id)->lockForUpdate()->first();
            $lockedTopup->update(['status' => 'successful']);
            $user->increment('wallet_balance', $lockedTopup->amount);

            Transaction::create(['user_id' => $user->id, 'type' => 'wallet_topup', 'amount' => $lockedTopup->amount, 'description' => 'Wallet top up via Paystack', 'reference' => $txRef, 'status' => 'successful']);
            $notifyUser = $user;
        });

        if ($notifyUser) {
            try {
                $notifyUser->notify(new \App\Notifications\WalletTopupCredited($amount));
            } catch (\Throwable $e) {
                \Illuminate\Support\Facades\Log::warning('WalletTopupCredited notify failed: ' . $e->getMessage());
            }
        }
    }

    private function creditWallet(string $paystackId, float $amount, ?string $accountNumber, ?string $email = null): void
    {
        $ref = 'WEBHOOK-' . $paystackId;

        $user = $accountNumber ? User::where('virtual_account_number', $accountNumber)->first() : null;
        if (!$user && $email) {
            $user = User::where('email', $email)->first();
        }
        if (!$user) return;

        $notifyUser = null;

        DB::transaction(function () use ($ref, $amount, $user, &$notifyUser) {
            $lockedUser = User::where('id', $user->id)->lockForUpdate()->first();
            try {
                WalletTopup::create(['user_id' => $lockedUser->id, 'tx_ref' => $ref, 'amount' => $amount, 'status' => 'successful']);
            } catch (\Illuminate\Database\QueryException $e) {
                return;
            }
            $lockedUser->increment('wallet_balance', $amount);
            Transaction::create(['user_id' => $lockedUser->id, 'type' => 'wallet_topup', 'amount' => $amount, 'description' => 'Wallet funded via Paystack', 'reference' => $ref, 'status' => 'successful']);
            $notifyUser = $lockedUser;
        });

        if ($notifyUser) {
            try {
                $notifyUser->notify(new \App\Notifications\WalletTopupCredited($amount));
            } catch (\Throwable $e) {
                \Illuminate\Support\Facades\Log::warning('WalletTopupCredited notify failed: ' . $e->getMessage());
            }
        }
    }
}

=== PaymentSettlementService ===
<?php

namespace App\Services;

use App\Models\Transaction;
use App\Models\User;
use App\Models\WalletTopup;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

/**
 * Single place where a verified Paystack payment turns into money/state changes.
 * Used by BOTH the browser callback and the webhook, so whichever arrives first
 * does the work and the other one is a safe no-op (row-locked + idempotent).
 * Each method refuses a reference with the wrong prefix, so a payment made for
 * one purpose can never be replayed against another.
 */
class PaymentSettlementService
{
    private function lockTopup(WalletTopup $topup, string $prefix, float $paidAmount): ?WalletTopup
    {
        $locked = WalletTopup::where('id', $topup->id)->lockForUpdate()->first();
        if (!$locked) return null;
        if (!str_starts_with((string) $locked->tx_ref, $prefix)) return null;
        if ($locked->status === 'successful') return null;
        if (round($paidAmount, 2) < round((float) $locked->amount, 2)) return null;
        return $locked;
    }

    public function settleWalletTopup(WalletTopup $topup, float $paidAmount): ?User
    {
        $notify = null;

        DB::transaction(function () use ($topup, $paidAmount, &$notify) {
            $locked = $this->lockTopup($topup, 'QELVO-', $paidAmount);
            if (!$locked) return;

            $user = User::where('id', $locked->user_id)->lockForUpdate()->first();
            if (!$user) return;

            $locked->update(['status' => 'successful']);
            $user->increment('wallet_balance', $locked->amount);

            Transaction::create([
                'user_id' => $user->id,
                'type' => 'wallet_topup',
                'amount' => $locked->amount,
                'description' => 'Wallet top up via Paystack',
            ]);
            $notify = $user;
        });

        if ($notify) {
            try {
                $notify->notify(new \App\Notifications\WalletTopupCredited((float) $topup->amount));
            } catch (\Throwable $e) {
                Log::warning('WalletTopupCredited notify failed: ' . $e->getMessage());
            }
        }

        return $notify;
    }

    /** Returns true only when this call newly activated the account. */
    public function settleSignupFee(WalletTopup $topup, float $paidAmount, array $paystackData = []): bool
    {
        $activated = false;

        DB::transaction(function () use ($topup, $paidAmount, $paystackData, &$activated) {
            $locked = $this->lockTopup($topup, 'SIGNUPFEE-', $paidAmount);
            if (!$locked) return;

            $user = User::where('id', $locked->user_id)->lockForUpdate()->first();
            if (!$user) return;

            $locked->update(['status' => 'successful']);
            if ($user->signup_fee_paid) return;

            $update = ['signup_fee_paid' => true, 'badge_tier' => 'bronze'];
            if ($user->tier === 'free' || $user->tier === null) {
                $update['tier'] = 'tier1';
                $update['tier_upgraded_at'] = now();
            }
            $auth = $paystackData['authorization'] ?? null;
            if ($auth && !empty($auth['authorization_code'])) {
                $update['card_token'] = $auth['authorization_code'];
                $update['card_last4'] = $auth['last4'] ?? null;
                $update['card_type'] = $auth['card_type'] ?? null;
            }
            $user->update($update);

            Transaction::create([
                'user_id' => $user->id,
                'type' => 'signup_fee',
                'amount' => -((float) $locked->amount),
                'description' => 'Account activation via Paystack',
            ]);
            $activated = true;
        });

        return $activated;
    }

    public function settleVendor(WalletTopup $topup, float $paidAmount): bool
    {
        $activated = false;

        DB::transaction(function () use ($topup, $paidAmount, &$activated) {
            $locked = $this->lockTopup($topup, 'VENDORREG-', $paidAmount);
            if (!$locked) return;

            $user = User::where('id', $locked->user_id)->lockForUpdate()->first();
            if (!$user) return;

            $fee = (float) $locked->amount;
            $locked->update(['status' => 'successful']);

            $update = ['is_vendor_active' => true];
            if ($user->tier === 'free' || $user->tier === null) {
                $update['tier'] = 'tier1';
                $update['tier_upgraded_at'] = now();
            }
            $user->update($update);

            Transaction::create([
                'user_id' => $user->id,
                'type' => 'vendor_registration',
                'amount' => -$fee,
                'description' => 'Vendor registration fee',
            ]);

            if ($user->referred_by) {
                $referrer = User::where('id', $user->referred_by)->lockForUpdate()->first();
                if ($referrer && !$referrer->wallet_restricted) {
                    $pct = in_array($referrer->tier, ['tier2', 'tier3'], true) ? 0.15 : 0;
                    $bonus = $fee * $pct;
                    if ($bonus > 0) {
                        $referrer->increment('wallet_balance', $bonus);
                        Transaction::create([
                            'user_id' => $referrer->id,
                            'type' => 'referral_bonus',
                            'amount' => $bonus,
                            'description' => 'Referral bonus from ' . $user->name . ' vendor registration',
                        ]);
                    }
                }
            }
            $activated = true;
        });

        return $activated;
    }
}

<?php

namespace App\Services;

use App\Models\Gig;
use App\Models\Transaction;
use App\Models\User;

class GigPayoutService
{
    /**
     * Pays a worker (and their referrer's bonus, if any) for a gig.
     * Must always be called from inside a DB::transaction() with $worker
     * already locked (lockForUpdate) by the caller — this method does not
     * open its own transaction or lock.
     *
     * $overrideAmount: if given, pays out based on this amount instead of
     * the gig's full price — used for partial dispute resolutions where
     * the worker is only paid for a percentage of the job. Null (default)
     * keeps the normal full-payout behavior.
     */
    public function payout(Gig $gig, User $worker, ?float $overrideAmount = null): void
    {
        $baseAmount = $overrideAmount ?? $gig->amount;

        $rate = (float) $worker->planRule('gigs_commission', 0.05);
        $commission = round($baseAmount * $rate, 2);
        $payout = $baseAmount - $commission;

        $worker->increment('wallet_balance', $payout);
        Transaction::create([
            'user_id' => $worker->id,
            'type' => 'gig_earning',
            'amount' => $payout,
            'description' => 'Completed gig: ' . $gig->title . ($commission > 0 ? ' (after ₦' . number_format($commission) . ' platform fee)' : '') . ($overrideAmount !== null ? ' (partial payout)' : ''),
        ]);
        \App\Models\PlatformRevenue::record('gigs', $commission, 'Commission: ' . $gig->title, 'gig', $gig->id);

        try {
            app(\App\Services\RankingService::class)->award($worker, 'gig_completed', null, 'gig:' . $gig->id);
        } catch (\Throwable $e) {
            report($e);
        }

        if ($worker->referred_by) {
            $referrer = User::where('id', $worker->referred_by)->lockForUpdate()->first();
            if ($referrer && !$referrer->wallet_restricted) {
                $pct = (float) $referrer->planRule('referral_pct', 0);
                $bonus = $payout * $pct;
                if ($bonus > 0) {
                    $referrer->increment('wallet_balance', $bonus);
                    Transaction::create([
                        'user_id' => $referrer->id,
                        'type' => 'referral_bonus',
                        'amount' => $bonus,
                        'description' => 'Referral bonus from ' . $worker->name . ' completing a gig',
                    ]);
                }
            }
        }
    }
}
