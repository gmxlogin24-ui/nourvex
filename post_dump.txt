<?php

namespace App\Http\Controllers;

use Illuminate\Support\Facades\Route;

class SeoController extends Controller
{
    public function robots()
    {
        $txt = "User-agent: *\nAllow: /\nDisallow: /admin\nDisallow: /api\n\nSitemap: " . url('/sitemap.xml') . "\n";
        return response($txt, 200, ['Content-Type' => 'text/plain; charset=UTF-8']);
    }

    public function sitemap()
    {
        $names = ['welcome', 'about', 'faq', 'safety', 'news.index', 'gigs.index', 'marketplace.index', 'pulse.index', 'register'];
        $urls = [];
        foreach ($names as $n) {
            if (Route::has($n)) {
                $urls[] = route($n);
            }
        }
        $now = now()->toAtomString();
        $xml = '<?xml version="1.0" encoding="UTF-8"?>' . "\n" . '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">' . "\n";
        foreach (array_unique($urls) as $u) {
            $xml .= '  <url><loc>' . e($u) . '</loc><lastmod>' . $now . '</lastmod></url>' . "\n";
        }
        $xml .= '</urlset>' . "\n";
        return response($xml, 200, ['Content-Type' => 'application/xml']);
    }
}

<?php

namespace App\Http\Controllers;

use App\Services\LevelBadgeService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\Rule;

class ProfileController extends Controller
{
    public function __construct(protected LevelBadgeService $levelBadgeService) {}

    public function index()
    {
        $user = Auth::user();

        $result = $this->levelBadgeService->refresh($user);
        $earnedBadges = $user->badges()->get();

        $allReferrals = \App\Models\User::where('referred_by', $user->id)->get();
        $activeReferrals = $allReferrals->where('tier', '!=', 'free');

        $recentReviews = \App\Models\Review::with('reviewer')
            ->where('reviewed_user_id', $user->id)
            ->latest()
            ->take(3)
            ->get();

        $myGigsPosted = \App\Models\Gig::where('user_id', $user->id)->count();
        $myListingsPosted = \App\Models\Listing::where('user_id', $user->id)->count();

        $monthStart = now()->startOfMonth();
        $monthIn = \App\Models\Transaction::where('user_id', $user->id)->where('amount', '>', 0)->where('created_at', '>=', $monthStart)->sum('amount');
        $monthOut = abs(\App\Models\Transaction::where('user_id', $user->id)->where('amount', '<', 0)->where('created_at', '>=', $monthStart)->sum('amount'));

        $leaderboardRank = null;
        if ($user->leaderboard_opt_in) {
            $weekStart = now()->startOfWeek();
            $rankedUsers = \App\Models\Transaction::where('transactions.created_at', '>=', $weekStart)
                ->where('amount', '>', 0)
                ->whereIn('type', ['gig_earning', 'marketplace_sale', 'referral_bonus'])
                ->join('users', 'transactions.user_id', '=', 'users.id')
                ->where('users.leaderboard_opt_in', true)
                ->selectRaw('users.id, SUM(transactions.amount) as total')
                ->groupBy('users.id')
                ->orderByDesc('total')
                ->pluck('users.id')
                ->values();
            $rankIndex = $rankedUsers->search($user->id);
            $leaderboardRank = $rankIndex === false ? null : $rankIndex + 1;
        }

        return view('profile.index', [
            'user' => $user,
            'workerLevel' => $user->workerLevel(),
            'completedGigsCount' => $user->completedGigsCount(),
            'averageRating' => $user->averageRatingReceived(),
            'reviewsReceivedCount' => $user->reviewsReceivedCount(),
            'currentLevel' => $result['level'],
            'earnedBadges' => $earnedBadges->take(6),
            'totalBadgesEarned' => $earnedBadges->count(),
            'allReferralsCount' => $allReferrals->count(),
            'activeReferralsCount' => $activeReferrals->count(),
            'recentReferrals' => $allReferrals->sortByDesc('created_at')->take(4)->values(),
            'recentReviews' => $recentReviews,
            'myGigsPosted' => $myGigsPosted,
            'myListingsPosted' => $myListingsPosted,
            'monthIn' => $monthIn,
            'monthOut' => $monthOut,
            'profileLeaderboardRank' => $leaderboardRank,
        ]);
    }

    public function update(Request $request)
    {
        \Log::info('PROFILE UPDATE RAW accent_color: ' . var_export($request->input('accent_color'), true));
        $user = Auth::user();

        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'email' => ['required', 'email', Rule::unique('users')->ignore($user->id)],
            'accent_color' => 'nullable|string|max:7',
            'theme_style' => 'nullable|string|in:glass,neumorphic,plain,aurora',
            'mobile_nav_style' => 'nullable|string|in:painted_pill,edge_dock,halo_dock,liquid_glass',
            'sidebar_style' => 'nullable|string|in:classic,rail',
            'avatar' => 'nullable|image|max:4096',
            'background' => 'nullable|image|max:8192',
        ]);

        if ($request->hasFile('avatar')) {
            $path = $request->file('avatar')->store('avatars', 'public');
            $validated['avatar_url'] = '/storage/' . $path;
        }

        if ($request->hasFile('background')) {
            $path = $request->file('background')->store('backgrounds', 'public');
            $validated['background_url'] = '/storage/' . $path;
        }

        unset($validated['avatar'], $validated['background']);

        $user->update($validated);

        return back()->with('status', 'Profile updated.');
    }

            $allListings = $query->get();

<?php

namespace App\Http\Controllers;

use App\Models\Listing;
use App\Models\ListingDispute;
use App\Models\Transaction;
use App\Models\User;
use App\Notifications\ListingDisputeFiled;
use App\Notifications\ListingDisputeResolved;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Notification;

class ListingDisputeController extends Controller
{
    // Buyer files a complaint. Allowed any time escrow is still held —
    // before confirming (item never arrived/wrong item) or after shipping
    // was marked but nothing showed up. Not for the seller to file (they
    // already got what they wanted the moment the buyer paid).
    public function store(Request $request, Listing $listing)
    {
        $user = Auth::user();
        abort_unless($listing->buyer_id === $user->id, 403, 'Only the buyer can file a complaint on this listing.');
        abort_unless($listing->status === 'sold', 422, 'This listing has no pending escrow to dispute.');

        $existing = ListingDispute::where('listing_id', $listing->id)->first();
        if ($existing && $existing->status === 'open') {
            return back()->withErrors(['dispute' => 'A complaint is already open for this listing.']);
        }

        $validated = $request->validate(['description' => 'required|string|max:2000']);

        $dispute = ListingDispute::updateOrCreate(
            ['listing_id' => $listing->id],
            [
                'raised_by' => $user->id,
                'description' => $validated['description'],
                'status' => 'open',
                'resolved_by' => null,
                'resolved_at' => null,
            ]
        );

        Notification::send(User::where('is_admin', true)->get(), new ListingDisputeFiled($dispute));

        if ($listing->user_id) {
            \App\Models\ListingMessage::postSystemEvent(
                $listing->id, $user->id, $listing->user_id,
                '⚠️ A complaint was filed on this listing — admin is reviewing.'
            );
        }

        return back()->with('status', 'Complaint submitted — admin will review it.');
    }

    public function index()
    {
        abort_unless(Auth::user()->is_admin, 403);
        $disputes = ListingDispute::where('status', 'open')->with(['listing', 'raisedBy', 'assignedTo'])->latest()->get();
        $ccAgents = User::where('role', 'customer_care')->orderBy('name')->get();
        return view('admin.marketplace-disputes.index', ['disputes' => $disputes, 'ccAgents' => $ccAgents]);
    }

    // Admin sides with the buyer: escrowed funds already sitting in nobody's
    // wallet get released to the buyer's wallet (refund). No commission taken.
    public function resolveRefunded(ListingDispute $dispute)
    {
        $admin = Auth::user();
        abort_unless($admin->is_admin, 403);
        abort_unless($dispute->status === 'open', 422, 'This complaint has already been resolved.');

        DB::transaction(function () use ($dispute, $admin) {
            $listing = Listing::where('id', $dispute->listing_id)->lockForUpdate()->first();
            abort_unless($listing->status === 'sold', 422, 'This listing is no longer awaiting payment resolution.');

            $buyer = User::where('id', $listing->buyer_id)->lockForUpdate()->first();

            $buyer->increment('wallet_balance', $listing->amount);
            Transaction::create([
                'user_id' => $buyer->id,
                'type' => 'marketplace_refund',
                'amount' => $listing->amount,
                'description' => 'Admin refund (dispute): ' . $listing->title,
            ]);

            $listing->update(['status' => 'available', 'buyer_id' => null]);

            $dispute->update(['status' => 'resolved_refunded', 'resolved_by' => $admin->id, 'resolved_at' => now()]);
        });

        $this->notifyResolution($dispute->fresh(), 'refunded');

        return back()->with('status', 'Dispute resolved — buyer has been refunded.');
    }

    // Admin sides with the seller: escrow releases normally, same commission
    // logic as a normal confirmed delivery.
    public function resolvePaid(ListingDispute $dispute)
    {
        $admin = Auth::user();
        abort_unless($admin->is_admin, 403);
        abort_unless($dispute->status === 'open', 422, 'This complaint has already been resolved.');

        DB::transaction(function () use ($dispute, $admin) {
            $listing = Listing::where('id', $dispute->listing_id)->lockForUpdate()->first();
            abort_unless($listing->status === 'sold', 422, 'This listing is no longer awaiting payment resolution.');

            $seller = $listing->user_id ? User::where('id', $listing->user_id)->lockForUpdate()->first() : null;

            if ($seller) {
                $rate = match ($seller->tier) {
                    'tier2', 'tier3' => 0,
                    'tier1' => 0.03,
                    default => 0.05,
                };
                $commission = round($listing->amount * $rate, 2);
                $payout = $listing->amount - $commission;

                $seller->increment('wallet_balance', $payout);
                Transaction::create([
                    'user_id' => $seller->id,
                    'type' => 'marketplace_sale',
                    'amount' => $payout,
                    'description' => 'Sold (dispute resolved): ' . $listing->title . ($commission > 0 ? ' (after ₦' . number_format($commission) . ' platform fee)' : ''),
                ]);
                \App\Models\PlatformRevenue::record('marketplace', $commission, 'Commission (dispute resolved): ' . $listing->title, 'listing', $listing->id);
            }

            $listing->update(['status' => 'completed']);

            $dispute->update(['status' => 'resolved_paid', 'resolved_by' => $admin->id, 'resolved_at' => now()]);
        });

        $this->notifyResolution($dispute->fresh(), 'paid');

        return back()->with('status', 'Dispute resolved — seller has been paid.');
    }

    public function dismiss(ListingDispute $dispute)
    {
        $admin = Auth::user();
        abort_unless($admin->is_admin, 403);
        abort_unless($dispute->status === 'open', 422, 'This complaint has already been resolved.');

        $dispute->update(['status' => 'dismissed', 'resolved_by' => $admin->id, 'resolved_at' => now()]);

        $this->notifyResolution($dispute->fresh(), 'dismissed');

<?php

namespace App\Http\Controllers;

use App\Models\News;
use App\Models\Transaction;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class NewsController extends Controller
{
    // Duration options in days => price in naira. tier1 (Plus) cannot post ads — requires tier2+.
    const AD_PRICING = [3 => 1500, 7 => 3000, 14 => 5000];

    public function index()
    {
        $newsItems = News::where('type', '!=', 'ad')
            ->orWhere(function ($q) {
                $q->where('type', 'ad')->where(function ($q2) {
                    $q2->whereNull('expires_at')->orWhere('expires_at', '>', now());
                });
            })
            ->orderByDesc('is_featured')
            ->latest('published_at')
            ->get();

        return view('news.index', ['newsItems' => $newsItems]);
    }

    public function createAd()
    {
        abort_if(Auth::user()->tier === 'tier1', 403, 'Ads aren\'t available on the Silver plan — upgrade to Gold to post one.');
        return view('news.create-ad', ['pricing' => self::AD_PRICING]);
    }

    public function storeAd(Request $request)
    {
        $user = Auth::user();
        abort_if($user->tier === 'tier1', 403, 'Ads aren\'t available on the Silver plan — upgrade to Gold to post one.');

        $validated = $request->validate([
            'title' => 'required|string|max:255',
            'category' => 'required|string|max:100',
            'body' => 'required|string',
            'duration' => 'required|integer|in:3,7,14',
            'image' => 'nullable|image|max:6144',
        ]);

        $cost = self::AD_PRICING[(int) $validated['duration']];

        $imageUrl = null;
        if ($request->hasFile('image')) {
            $imageUrl = '/storage/' . $request->file('image')->store('news', 'public');
        }

        // Balance check + decrement + ad creation must be one atomic unit,
        // otherwise concurrent submissions can both pass the balance check
        // before either decrements (race condition -> negative balance).
        $news = DB::transaction(function () use ($user, $validated, $cost, $imageUrl) {
            $lockedUser = User::where('id', $user->id)->lockForUpdate()->first();

            if ($lockedUser->wallet_restricted) {
                return null;
            }
            if ($lockedUser->wallet_balance < $cost) {
                return null;
            }

            $lockedUser->decrement('wallet_balance', $cost);

            Transaction::create([
                'user_id' => $lockedUser->id,
                'type' => 'ad_post',
                'amount' => -$cost,
                'description' => $validated['duration'] . '-day ad: ' . $validated['title'],
            ]);

            return News::create([
                'title' => $validated['title'],
                'category' => $validated['category'],
                'body' => $validated['body'],
                'image_url' => $imageUrl,
                'type' => 'ad',
                'created_by' => $lockedUser->id,
                'published_at' => now(),
                'expires_at' => now()->addDays((int) $validated['duration']),
                'is_featured' => $lockedUser->tier === 'tier3',
            ]);
        });

        if (!$news) {
            $user->refresh();
            if ($user->wallet_restricted) {
                return back()->withErrors(['balance' => 'Your wallet is restricted due to an unresolved payment issue. Clear your balance to post ads.'])->withInput();
            }
            return back()->withErrors(['balance' => 'Insufficient wallet balance. This ad costs ₦' . number_format($cost) . '. Top up your wallet first.'])->withInput();
        }

        return redirect()->route('news.index')->with('status', 'Ad is live for ' . $validated['duration'] . ' days.');
    }

    public function store(Request $request)
    {
        abort_unless(Auth::user()->is_admin, 403);

        $validated = $request->validate([
            'title' => 'required|string|max:255',
            'category' => 'required|string|max:100',
            'body' => 'required|string',
            'type' => 'nullable|in:news,ad',
            'source_url' => 'nullable|url',
            'image' => 'nullable|image|max:6144',
            'video' => 'nullable|mimetypes:video/mp4,video/quicktime|max:20480',
        ]);

        if ($request->hasFile('image')) {
            $validated['image_url'] = '/storage/' . $request->file('image')->store('news', 'public');
        }
        if ($request->hasFile('video')) {
            $validated['video_url'] = '/storage/' . $request->file('video')->store('news', 'public');
        }
        unset($validated['image'], $validated['video']);

        News::create([
            ...$validated,
            'type' => $validated['type'] ?? 'news',
            'created_by' => Auth::id(),
            'published_at' => now(),
        ]);

        return redirect()->route('news.index');
    }
}

<?php

namespace App\Models;

trait HasPlanRules
{
    public function planKey(): string
    {
        $tier = $this->tier ?? 'free';
        return config('plans.' . $tier) ? $tier : 'free';
    }

    public function planLabel(): string
    {
        return Subscription::PLANS[$this->planKey()]['label'] ?? 'Free';
    }

    public function planRule(string $key, $default = null)
    {
        return config('plans.' . $this->planKey() . '.' . $key, $default);
    }

    // Use in Blade: @if(auth()->user()->planAllows('gigs_post'))
    public function planAllows(string $key): bool
    {
        if ($this->isAdmin()) {
            return true;
        }
        $v = $this->planRule($key, false);
        if ($v === null) {
            return true;
        }
        if (is_bool($v)) {
            return $v;
        }
        if (is_numeric($v)) {
            return $v > 0;
        }
        return (bool) $v;
    }
}
