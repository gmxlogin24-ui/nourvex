<?php

namespace App\Services;

use App\Models\QCoinTransaction;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use RuntimeException;

class QCoinService
{
    // Kept for older code that may still reference it. Daily claims now come from
    // config/plans.php (qcoin_daily) plus the personal streak bonus (config/qcoin.php).
    public const DAILY_CLAIM_AMOUNT = 50;

    /**
     * Credit Q Coins to a user. Atomic: row-locks the user, updates balance,
     * writes the ledger entry, all inside one DB transaction. This is the
     * ONLY method in the codebase that should increase q_coin_balance.
     */
    public function earn(User $user, int $amount, string $type, ?string $description = null, ?string $referenceId = null): QCoinTransaction
    {
        if ($amount <= 0) {
            throw new RuntimeException('Earn amount must be positive.');
        }

        return DB::transaction(function () use ($user, $amount, $type, $description, $referenceId) {
            $locked = User::whereKey($user->id)->lockForUpdate()->firstOrFail();

            $locked->q_coin_balance += $amount;
            $locked->save();

            $txn = QCoinTransaction::create([
                'user_id' => $locked->id,
                'amount' => $amount,
                'balance_after' => $locked->q_coin_balance,
                'type' => $type,
                'description' => $description,
                'reference_id' => $referenceId,
            ]);

            // Keep the caller's in-memory model in sync.
            $user->q_coin_balance = $locked->q_coin_balance;

            return $txn;
        });
    }

    /**
     * Debit Q Coins from a user (spend/redemption). Throws if insufficient balance.
     * ONLY method that should decrease q_coin_balance.
     */
    public function spend(User $user, int $amount, string $type, ?string $description = null, ?string $referenceId = null): QCoinTransaction
    {
        if ($amount <= 0) {
            throw new RuntimeException('Spend amount must be positive.');
        }

        return DB::transaction(function () use ($user, $amount, $type, $description, $referenceId) {
            $locked = User::whereKey($user->id)->lockForUpdate()->firstOrFail();

            if ($locked->q_coin_balance < $amount) {
                throw new RuntimeException('Insufficient Q Coin balance.');
            }

            $locked->q_coin_balance -= $amount;
            $locked->save();

            $txn = QCoinTransaction::create([
                'user_id' => $locked->id,
                'amount' => -$amount,
                'balance_after' => $locked->q_coin_balance,
                'type' => $type,
                'description' => $description,
                'reference_id' => $referenceId,
            ]);

            $user->q_coin_balance = $locked->q_coin_balance;

            return $txn;
        });
    }

    // Base daily claim for the user's plan (admins get the top plan's amount).
    public function baseFor(User $user): int
    {
        if ($user->isAdmin()) {
            return (int) config('plans.tier3.qcoin_daily', 0);
        }
        return (int) $user->planRule('qcoin_daily', 0);
    }

    // Personal streak bonus: grows by step_bonus every step_days consecutive days, capped.
    public function streakBonusFor(int $streakDays): int
    {
        $stepDays = max(1, (int) config('qcoin.streak_step_days', 7));
        $stepBonus = (int) config('qcoin.streak_step_bonus', 5);
        $max = (int) config('qcoin.streak_max_bonus', 25);

        return min(intdiv(max($streakDays, 0), $stepDays) * $stepBonus, $max);
    }

    /**
     * Daily claim, with the "already claimed today" check done INSIDE the
     * lock so two simultaneous requests can't both succeed.
     */
    public function claimDaily(User $user): QCoinTransaction
    {
        $today = now()->toDateString();
        $yesterday = now()->subDay()->toDateString();

        return DB::transaction(function () use ($user, $today, $yesterday) {
            $locked = User::whereKey($user->id)->lockForUpdate()->firstOrFail();

            if ($locked->q_coin_last_claim_date === $today) {
                throw new RuntimeException('Already claimed today.');
            }

            $base = $this->baseFor($locked);
            if ($base <= 0) {
                throw new RuntimeException('Your plan does not include daily Q Coins.');
            }

            $newStreak = ($locked->q_coin_last_claim_date === $yesterday)
                ? $locked->q_coin_streak_days + 1
                : 1;

            $bonus = $this->streakBonusFor($newStreak);
            $total = $base + $bonus;

            $locked->q_coin_balance += $total;
            $locked->q_coin_last_claim_date = $today;
            $locked->q_coin_streak_days = $newStreak;
            $locked->save();

            $txn = QCoinTransaction::create([
                'user_id' => $locked->id,
                'amount' => $total,
                'balance_after' => $locked->q_coin_balance,
                'type' => 'daily_claim',
                'description' => $bonus > 0
                    ? "Daily claim + streak bonus (day {$newStreak})"
                    : 'Daily Q Coin claim',
            ]);

            $user->q_coin_balance = $locked->q_coin_balance;
            $user->q_coin_last_claim_date = $today;
            $user->q_coin_streak_days = $newStreak;

            return $txn;
        });
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Transaction;
use App\Models\User;
use App\Models\Withdrawal;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Http;

class WalletActionController extends Controller
{
    // No-BVN accounts pay a 1.5% withdrawal fee; verified (virtual account / BVN)
    // accounts withdraw free. This is the only friction non-BVN users face -
    // withdrawals are never blocked, just costlier, matching the deposit cap logic.
    const NO_BVN_WITHDRAWAL_FEE_RATE = 0.015;

    public function withdraw(Request $request)
    {
        $validated = $request->validate([
            'amount' => 'required|numeric|min:100',
            'pin' => 'required|string',
        ]);
        $user = Auth::user();

        if (!$user->hasTransactionPin()) {
            return back()->withErrors(['pin' => 'Set a transaction PIN before withdrawing.']);
        }
        if (!$user->verifyTransactionPin($validated['pin'])) {
            return back()->withErrors(['pin' => 'Incorrect transaction PIN.']);
        }
        if (!$user->payout_account_number) {
            return back()->withErrors(['amount' => 'Add a payout bank account before withdrawing.']);
        }

        $hasVirtualAccount = (bool) $user->virtual_account_number;
        $fee = $hasVirtualAccount ? 0 : round($validated['amount'] * self::NO_BVN_WITHDRAWAL_FEE_RATE, 2);
        $totalDebit = $validated['amount'] + $fee;
        $txRef = 'WD-' . $user->id . '-' . time();

        // Debit immediately so the funds can't be double-spent while the
        // transfer is in flight. If the Paystack call fails to even
        // initiate, we reverse this below. If it initiates but the bank
        // payout later fails, the transfer.completed webhook reverses it.
        $debited = DB::transaction(function () use ($user, $validated, $fee, $totalDebit, $txRef) {
            $lockedUser = User::where('id', $user->id)->lockForUpdate()->first();

            if ($lockedUser->wallet_restricted) {
                return false;
            }

            if ($lockedUser->wallet_balance < $totalDebit) {
                return false;
            }

            $lockedUser->decrement('wallet_balance', $totalDebit);

            Withdrawal::create([
                'user_id' => $lockedUser->id,
                'tx_ref' => $txRef,
                'amount' => $validated['amount'],
                'fee' => $fee,
                'bank_code' => $lockedUser->payout_bank_code,
                'bank_name' => $lockedUser->payout_bank_name,
                'account_number' => $lockedUser->payout_account_number,
                'account_name' => $lockedUser->payout_account_name,
                'status' => 'pending',
            ]);

            Transaction::create([
                'user_id' => $lockedUser->id,
                'type' => 'withdrawal',
                'amount' => -$validated['amount'],
                'description' => 'Withdrawal to ' . $lockedUser->payout_bank_name . ' ****' . substr($lockedUser->payout_account_number, -4),
            ]);

            if ($fee > 0) {
                Transaction::create([
                    'user_id' => $lockedUser->id,
                    'type' => 'withdrawal_fee',
                    'amount' => -$fee,
                    'description' => 'No-BVN withdrawal fee (1.5%)',
                ]);
            }

            return true;
        });

        if (!$debited) {
            $restricted = User::where('id', $user->id)->value('wallet_restricted');
            if ($restricted) {
                return back()->withErrors(['amount' => 'Your wallet is restricted due to an unresolved payment issue. Clear your balance to make withdrawals.']);
            }
            $msg = $fee > 0
                ? 'Insufficient balance. You need ₦' . number_format($totalDebit, 2) . ' (amount + 1.5% no-BVN fee).'
                : 'Insufficient balance.';
            return back()->withErrors(['amount' => $msg]);
        }

        try {
            $secret = config('services.paystack.secret');

            // Paystack requires a transfer recipient to exist before you can pay it.
            $recipientRes = Http::withToken($secret)->post('https://api.paystack.co/transferrecipient', [
                'type' => 'nuban',
                'name' => $user->payout_account_name,
                'account_number' => $user->payout_account_number,
                'bank_code' => $user->payout_bank_code,
                'currency' => 'NGN',
            ]);
            $recipientData = $recipientRes->json();
            $recipientCode = $recipientData['data']['recipient_code'] ?? null;

            if (!$recipientCode) {
                $initiated = false;
            } else {
                $response = Http::withToken($secret)->post('https://api.paystack.co/transfer', [
                    'source' => 'balance',
                    'amount' => (int) round($validated['amount'] * 100),
                    'recipient' => $recipientCode,
                    'reference' => $txRef,
                    'reason' => 'NAURVEX withdrawal',
                ]);
                $data = $response->json();
                $initiated = ($data['status'] ?? false) === true && !in_array($data['data']['status'] ?? '', ['otp', 'failed', 'reversed'], true);
            }
        } catch (\Throwable $e) {
            \Illuminate\Support\Facades\Log::error('Withdrawal transfer init failed: ' . $e->getMessage());
            $initiated = false;
        }

        if (!$initiated) {
            $this->reverseWithdrawal($txRef, 'Withdrawal could not be started - refunded to wallet');
            return back()->withErrors(['amount' => 'Could not start the transfer. Your wallet has not been charged.']);
        }

        return redirect()->route('wallet.index')->with('status', 'Withdrawal of ₦' . number_format($validated['amount'], 2) . ' is processing.');
    }

    public function reverseWithdrawal(string $txRef, string $reason): void
    {
        DB::transaction(function () use ($txRef, $reason) {
            $withdrawal = Withdrawal::where('tx_ref', $txRef)->lockForUpdate()->first();
            if (!$withdrawal || $withdrawal->status !== 'pending') {
                return;
            }

            $withdrawal->update(['status' => 'failed']);
            $user = User::where('id', $withdrawal->user_id)->lockForUpdate()->first();
            $refundAmount = (float) $withdrawal->amount + (float) $withdrawal->fee;
            $user->increment('wallet_balance', $refundAmount);

            Transaction::create([
                'user_id' => $user->id,
                'type' => 'withdrawal_reversal',
                'amount' => $refundAmount,
                'description' => $reason,
            ]);
        });
    }
}

namespace App\Http\Controllers;

use App\Models\Transaction;

use Illuminate\Http\Request;

use Illuminate\Support\Facades\DB;

class WalletController extends Controller
{
    public function sendMoneyForm()
    {
        return view('wallet.send', [
            'hasPin' => Auth::user()->hasTransactionPin(),
        ]);
    }

    // AJAX: show the sender the recipient's real name before they commit to
    // sending. Display-only - sendMoney() below re-looks-up the recipient
    // itself before crediting anything, so this can't be tampered with
    // client-side, matching BankAccountController::resolve()'s pattern.
    public function resolveRecipient(Request $request)
    {
        $validated = $request->validate([
            'recipient_identifier' => 'required|string',
        ]);

        $sender = Auth::user();
        $recipient = User::where('email', $validated['recipient_identifier'])
            ->orWhere('id', $validated['recipient_identifier'])
            ->first();

        if (!$recipient) {
            return response()->json(['error' => 'No user found with that email or ID.'], 422);
        }
        if ($recipient->id === $sender->id) {
            return response()->json(['error' => 'You cannot send money to yourself.'], 422);
        }

        return response()->json(['name' => $recipient->name]);
    }

    // Locked transaction, mirrors the pattern used everywhere else in this app
    // (GigController deposit, NewsController::storeAd, ServicesController::processPurchase)
    // to prevent race conditions on concurrent transfer requests.
    public function sendMoney(Request $request)
    {
        $sender = Auth::user();
        $validated = $request->validate([
            'recipient_identifier' => 'required|string',
            'amount' => 'required|numeric|min:1',
            'note' => 'nullable|string|max:255',
            'pin' => 'required|string',
        ]);

        if (!$sender->hasTransactionPin()) {
            return back()->withErrors(['pin' => 'Set a transaction PIN before sending money.'])->withInput();
        }
        if (!$sender->verifyTransactionPin($validated['pin'])) {
            return back()->withErrors(['pin' => 'Incorrect transaction PIN.'])->withInput();
        }

        $recipient = User::where('email', $validated['recipient_identifier'])
            ->orWhere('id', $validated['recipient_identifier'])
            ->first();

        if (!$recipient) {
            return back()->withErrors(['recipient_identifier' => 'No user found with that email or ID.'])->withInput();
        }

        if ($recipient->id === $sender->id) {
            return back()->withErrors(['recipient_identifier' => 'You cannot send money to yourself.'])->withInput();
        }

        $result = DB::transaction(function () use ($sender, $recipient, $validated) {
            $lockedSender = User::where('id', $sender->id)->lockForUpdate()->first();

            if ($lockedSender->wallet_restricted) {
                return ['status' => 'restricted'];
            }
            if ($lockedSender->wallet_balance < $validated['amount']) {
                return ['status' => 'insufficient'];
            }

            $lockedRecipient = User::where('id', $recipient->id)->lockForUpdate()->first();

            if ($lockedRecipient->wouldExceedNoBvnCap($validated['amount'])) {
                return ['status' => 'recipient_capped'];
            }

            $lockedSender->decrement('wallet_balance', $validated['amount']);
            $lockedRecipient->increment('wallet_balance', $validated['amount']);

            $note = $validated['note'] ?? '';

            // Internal wallet-to-wallet transfer - no bank/Paystack is involved,
            // so NAURVEX generates its own reference, shared by both legs of
            // the transfer (same convention banks use between debit/credit legs).
            $reference = 'NRX' . now()->format('ymd') . strtoupper(substr(bin2hex(random_bytes(4)), 0, 6));

            $senderTxn = Transaction::create([
                'user_id' => $lockedSender->id,
                'related_user_id' => $lockedRecipient->id,
                'type' => 'transfer_sent',
                'amount' => -$validated['amount'],
                'description' => 'Sent to ' . $lockedRecipient->name . ($note ? ': ' . $note : ''),
                'reference' => $reference,
                'status' => 'successful',
            ]);
            Transaction::create([
                'user_id' => $lockedRecipient->id,
                'related_user_id' => $lockedSender->id,
                'type' => 'transfer_received',
                'amount' => $validated['amount'],
                'description' => 'Received from ' . $lockedSender->name . ($note ? ': ' . $note : ''),
                'reference' => $reference,
                'status' => 'successful',
            ]);
            $lockedRecipient->notify(new \App\Notifications\MoneyReceived($lockedSender, $validated['amount'], $note ?: null));

            return ['status' => 'ok', 'txn_id' => $senderTxn->id];
        });

        return match ($result['status']) {
            'restricted' => back()->withErrors(['amount' => 'Your wallet is restricted due to an unresolved payment issue.'])->withInput(),
            'insufficient' => back()->withErrors(['amount' => 'Insufficient wallet balance.'])->withInput(),
            'recipient_capped' => back()->withErrors(['recipient_identifier' => 'This recipient has an unverified wallet and cannot receive that much right now.'])->withInput(),
            default => redirect()->route('wallet.transaction.show', $result['txn_id']),
        };
    }

    public function moneyTracker()
    {
        $user = Auth::user();

        abort_unless(in_array($user->tier, ['tier2', 'tier3'], true), 403, 'Money Tracker is available on the Gold plan and above. Upgrade to unlock detailed cash flow tracking.');

        $totalIn = Transaction::where('user_id', $user->id)->where('amount', '>', 0)->sum('amount');
        $totalOut = abs(Transaction::where('user_id', $user->id)->where('amount', '<', 0)->sum('amount'));

        $categoryTotals = Transaction::where('user_id', $user->id)
            ->selectRaw('type, SUM(ABS(amount)) as total')
            ->groupBy('type')
            ->pluck('total', 'type');

        $driver = \DB::connection()->getDriverName();
        $ymExpr = match ($driver) {
            'pgsql' => "TO_CHAR(created_at, 'YYYY-MM')",
            'sqlite' => "strftime('%Y-%m', created_at)",
            default => "DATE_FORMAT(created_at, '%Y-%m')", // mysql/mariadb
        };

        $monthly = Transaction::where('user_id', $user->id)
            ->where('created_at', '>=', now()->subMonths(6))
            ->selectRaw("$ymExpr as ym, SUM(CASE WHEN amount > 0 THEN amount ELSE 0 END) as inflow, SUM(CASE WHEN amount < 0 THEN ABS(amount) ELSE 0 END) as outflow")
            ->groupBy('ym')
            ->orderBy('ym')
            ->get();

        return view('money-tracker.index', [
            'balance' => $user->wallet_balance,
            'totalIn' => $totalIn,
            'totalOut' => $totalOut,
            'categoryTotals' => $categoryTotals,
            'monthly' => $monthly,
            'transactions' => Transaction::where('user_id', $user->id)->latest()->take(30)->get(),
        ]);
    }

    public function deposit()
    {
        $user = Auth::user();

        return view('wallet.deposit', [
            'user' => $user,
            'balance' => $user->wallet_balance,
            'hasVirtual' => (bool) $user->virtual_account_number,
            'walletCapRemaining' => $user->virtual_account_number ? null : max(0, \App\Models\User::NO_BVN_WALLET_CAP - $user->wallet_balance),
            'recent' => \App\Models\Transaction::where('user_id', $user->id)->where('type', 'wallet_topup')->latest()->limit(5)->get(),
        ]);
    }

    public function depositStatus($reference)
    {
        $topup = \App\Models\WalletTopup::where('tx_ref', $reference)->where('user_id', Auth::id())->firstOrFail();
        $tx = $topup->status === 'successful'
            ? \App\Models\Transaction::where('user_id', Auth::id())->where('type', 'wallet_topup')->where('amount', $topup->amount)->latest()->first()
            : null;

        return view('wallet.deposit-status', [
            'topup' => $topup,
            'tx' => $tx,
            'balance' => Auth::user()->wallet_balance,
        ]);
    }
}

class AdminCoinController extends Controller
{
    private function safe(callable $f)
    {
        try { return $f(); } catch (\Throwable $e) { return null; }
    }

    public function index()
    {
        $rate = 20.0;
        foreach (['qcoin.rate', 'qcoin.display_rate', 'qcoin.per_naira', 'qcoin.q_per_naira', 'investment.qcoin_per_naira'] as $k) {
            $v = config($k);
            if (is_numeric($v) && $v > 0) { $rate = (float) $v; break; }
        }
        $daily = [];
        $fees = [];
        $defD = [0, 10, 15, 20];
        $defF = [0.05, 0.03, 0, 0];
        for ($t = 0; $t < 4; $t++) {
            $daily[] = (float) config("plans.rules.tier$t.qcoin_daily", config("plans.tier$t.qcoin_daily", $defD[$t]));
            $fees[] = (float) config("plans.rules.tier$t.gigs_commission", config("plans.tier$t.gigs_commission", $defF[$t]));

        $tiers = $this->safe(fn () => DB::table('users')->selectRaw('tier, count(*) as c')->groupBy('tier')->pluck('c', 'tier')->all()) ?: [];
        $day = now()->subDay();
        $data = [
            'rate' => $rate,
            'users' => (int) $this->safe(fn () => DB::table('users')->count()),
            'tiers' => $tiers,
            'daily' => $daily,
            'fees' => $fees,
            'platformFee' => (float) config('investment.platform_fee_percent', 5),

            'holders' => $hasBal ? (int) $this->safe(fn () => DB::table('users')->where('q_coin_balance', '>', 0)->count()) : 0,
            'tx' => $hasTx ? (int) $this->safe(fn () => DB::table('q_coin_transactions')->count()) : 0,
            'tx24' => $hasTx ? (int) $this->safe(fn () => DB::table('q_coin_transactions')->where('created_at', '>=', $day)->count()) : 0,
            'in24' => $hasTx ? (float) $this->safe(fn () => DB::table('q_coin_transactions')->where('created_at', '>=', $day)->where('amount', '>', 0)->sum('amount')) : 0,

<?php

namespace App\Services;

use App\Models\Gig;
use App\Models\Transaction;
use App\Models\User;

class GigPayoutService
{
    /**
     * Pays a worker (and their referrer's bonus, if any) for a gig.
     * Must always be called from inside a DB::transaction() with $worker
     * already locked (lockForUpdate) by the caller — this method does not
     * open its own transaction or lock.
     *
     * $overrideAmount: if given, pays out based on this amount instead of
     * the gig's full price — used for partial dispute resolutions where
     * the worker is only paid for a percentage of the job. Null (default)
     * keeps the normal full-payout behavior.
     */
    public function payout(Gig $gig, User $worker, ?float $overrideAmount = null): void
    {
        $baseAmount = $overrideAmount ?? $gig->amount;

        $rate = (float) $worker->planRule('gigs_commission', 0.05);
        $commission = round($baseAmount * $rate, 2);
        $payout = $baseAmount - $commission;

        $worker->increment('wallet_balance', $payout);
        Transaction::create([
            'user_id' => $worker->id,
            'type' => 'gig_earning',
            'amount' => $payout,
            'description' => 'Completed gig: ' . $gig->title . ($commission > 0 ? ' (after ₦' . number_format($commission) . ' platform fee)' : '') . ($overrideAmount !== null ? ' (partial payout)' : ''),
        ]);
        \App\Models\PlatformRevenue::record('gigs', $commission, 'Commission: ' . $gig->title, 'gig', $gig->id);

        try {
            app(\App\Services\RankingService::class)->award($worker, 'gig_completed', null, 'gig:' . $gig->id);
        } catch (\Throwable $e) {
            report($e);
        }

        if ($worker->referred_by) {
            $referrer = User::where('id', $worker->referred_by)->lockForUpdate()->first();
            if ($referrer && !$referrer->wallet_restricted) {
                $pct = (float) $referrer->planRule('referral_pct', 0);
                $bonus = $payout * $pct;
                if ($bonus > 0) {
                    $referrer->increment('wallet_balance', $bonus);
                    Transaction::create([
                        'user_id' => $referrer->id,
                        'type' => 'referral_bonus',
                        'amount' => $bonus,
                        'description' => 'Referral bonus from ' . $worker->name . ' completing a gig',
                    ]);
                }
            }
        }
    }
}
